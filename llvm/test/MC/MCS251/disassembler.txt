#--- roundtrip.ll
target triple = "mcs251-unknown-none"

declare i16 @helper(i16)

define i16 @add_and_call(i16 %x) {
  %a = add i16 %x, 7
  %r = call i16 @helper(i16 %a)
  ret i16 %r
}

!mcs251.signatures = !{!0, !1}
!0 = !{!"_helper", i32 2, i32 0, i32 0}
!1 = !{!"_add_and_call", i32 1, i32 0, i32 0}
